SELECT COUNT(*) FROM patients;

SELECT COUNT(*) FROM conditions;

SELECT COUNT(*) FROM encounters e
LEFT JOIN patients p ON e.patient = p.id
WHERE p.id IS NULL;

SELECT COUNT(*) FROM conditions c
LEFT JOIN patients p ON c.patient = p.id
WHERE p.id IS NULL;

SELECT COUNT(*) FROM encounters WHERE stop_ts < start_ts;
SELECT COUNT(*) FROM conditions WHERE stop_date < start_date;

SELECT
  COUNT(*) FILTER (WHERE birthdate IS NULL) AS no_birthdate,
  COUNT(*) FILTER (WHERE gender IS NULL) AS no_gender,
  COUNT(*) FILTER (WHERE deathdate IS NOT NULL) AS deceased
FROM patients;

SELECT encounterclass, COUNT(*) 
FROM encounters
GROUP BY encounterclass
ORDER BY 2 DESC;

SELECT description, COUNT(*) AS n
FROM conditions
GROUP BY description
ORDER BY n DESC
LIMIT 25;
